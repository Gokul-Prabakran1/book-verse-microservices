# ----------- Stage 1: Base (Dependency Install) -----------
FROM node:20-alpine AS deps

WORKDIR /app

# Copy only package files and install dependencies
COPY package*.json ./
RUN npm install

# ----------- Stage 2: Dev Runtime -----------
FROM node:20-alpine AS dev

WORKDIR /app

# Copy node_modules from previous stage
COPY --from=deps /app/node_modules ./node_modules

# Copy source files
COPY . .

# Expose Vite dev server port
EXPOSE 5173

# Vite must bind to 0.0.0.0 inside Docker
ENV HOST=0.0.0.0

# Start dev server
CMD ["npm", "run", "dev"]
